
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(roles DBSvr LoginSvr GameSvr LoginGate SelGate GameGate)
get_target_property(common_sources Def SOURCES)
set(role_index 0)
set(core_libraries "")
foreach(role IN LISTS roles)
    set(core_sources "")
    math(EXPR role_index "${role_index} + 1")
    math(EXPR string_base "${role_index} * 8192")
    set(all_sources "")
    foreach(source IN LISTS ${role}_SERVICE_SOURCES)
        list(APPEND all_sources "${CMAKE_CURRENT_SOURCE_DIR}/../${role}/${source}")
    endforeach()
    foreach(source IN LISTS common_sources)
        if(source STREQUAL "ModernUI.cpp")
            continue()
        endif()
        list(APPEND all_sources "${CMAKE_CURRENT_SOURCE_DIR}/../Def/${source}")
    endforeach()
    set(source_index 0)
    foreach(source IN LISTS all_sources)
        math(EXPR source_index "${source_index} + 1")
        set(wrapper "${CMAKE_CURRENT_BINARY_DIR}/${role}_${source_index}.cpp")
        file(RELATIVE_PATH relative_source "${CMAKE_CURRENT_SOURCE_DIR}" "${source}")
        set(source_headers "")
        if(relative_source STREQUAL "../GameSvr/ProcessMonster.cpp")
            set(source_headers [=[
#include <future>
#pragma push_macro("min")
#pragma push_macro("max")
#undef min
#undef max
#define PICOJSON_USE_INT64
#include "../Def/picojson.h"
#include "../Def/JsonRpc.h"
#pragma pop_macro("max")
#pragma pop_macro("min")
]=])
        endif()
        configure_file(Core.cpp.in "${wrapper}" @ONLY)
        list(APPEND core_sources "${wrapper}")
    endforeach()
    add_library(${role} STATIC ${core_sources})
    target_include_directories(${role} PRIVATE . .. ../Def ../Def/_OrzEx)
    target_compile_definitions(${role} PRIVATE UNICODE _UNICODE MIR3_EISERVER)
    list(APPEND core_libraries ${role})
    add_executable(${role}Window WIN32 EXCLUDE_FROM_ALL ServiceWindow.cpp "${CMAKE_CURRENT_BINARY_DIR}/Strings.rc")
    set_target_properties(${role}Window PROPERTIES OUTPUT_NAME ${role})
    if(NOT MIR3_BUILD_EISERVER)
        set_property(TARGET ${role}Window PROPERTY EXCLUDE_FROM_ALL FALSE)
    endif()
    target_compile_definitions(${role}Window PRIVATE UNICODE _UNICODE SERVICE_ROLE=${role})
    target_include_directories(${role}Window PRIVATE .)
    target_link_libraries(${role}Window PRIVATE ${role} mir_sqlite ws2_32 advapi32 odbc32 odbccp32 comctl32)

endforeach()
execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/resources.py" "${CMAKE_CURRENT_BINARY_DIR}/Strings.rc" COMMAND_ERROR_IS_FATAL ANY)

set(core_libraries ${core_libraries} PARENT_SCOPE)
set(MIR3_SERVICE_STRINGS "${CMAKE_CURRENT_BINARY_DIR}/Strings.rc" PARENT_SCOPE)
