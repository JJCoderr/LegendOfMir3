cmake_minimum_required(VERSION 3.24)
project(mir3_client LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(MIR3_BUILD_GUI "Build the Windows Godot client" ON)
set(MIR3_RESOURCE_ROOT "" CACHE PATH "Original packed client resource directory (required for GUI)")
set(MIR3_GODOT_CPP "${CMAKE_CURRENT_SOURCE_DIR}/../../mir3_1.45/third_party/godot-cpp" CACHE PATH "godot-cpp source checkout")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/.gdignore" "")
add_library(mir3_ui STATIC src/resources/wil_archive.cpp src/resources/messages.cpp)
target_include_directories(mir3_ui PUBLIC src)
if(MSVC)
    target_compile_options(mir3_ui PUBLIC /utf-8)
endif()
if(MIR3_BUILD_GUI)
    if(NOT IS_DIRECTORY "${MIR3_RESOURCE_ROOT}/Data")
        message(FATAL_ERROR "Set MIR3_RESOURCE_ROOT to the original client directory containing Data; resources are never copied")
    endif()
    if(NOT WIN32)
        message(FATAL_ERROR "The legacy GDI renderer currently requires Windows")
    endif()
    set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/godot_profile.json" CACHE FILEPATH "" FORCE)
    add_subdirectory("${MIR3_GODOT_CPP}" godot-cpp EXCLUDE_FROM_ALL)
    add_library(mir3_text STATIC src/text/native_text.cpp src/chat/chat_history.cpp src/chat/chat_input.cpp)
    target_include_directories(mir3_text PUBLIC src)
    target_link_libraries(mir3_text PUBLIC gdi32 user32)
    add_library(mir3_client SHARED src/godot/register_types.cpp src/godot/hud_view.cpp src/godot/chat_input_view.cpp)
    target_link_libraries(mir3_client PRIVATE mir3_ui mir3_text godot-cpp)
    set_target_properties(mir3_client PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_CURRENT_SOURCE_DIR}/godot/bin>")
    configure_file(godot/resource_path.cfg.in "${CMAKE_CURRENT_SOURCE_DIR}/godot/resource_path.cfg" @ONLY)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(hud_checks tests/hud_checks.cpp)
    target_link_libraries(hud_checks PRIVATE mir3_ui)
    add_test(NAME hud_checks COMMAND hud_checks "${MIR3_RESOURCE_ROOT}")
    if(MIR3_BUILD_GUI)
        add_executable(text_checks tests/text_checks.cpp)
        target_link_libraries(text_checks PRIVATE mir3_text)
        if(MSVC)
            target_compile_options(text_checks PRIVATE /utf-8)
        endif()
        add_test(NAME text_checks COMMAND text_checks)
        find_program(MIR3_GODOT_EXECUTABLE NAMES Godot_v4.7.2-stable_win64_console.exe godot.exe godot)
        if(MIR3_GODOT_EXECUTABLE)
            add_test(NAME godot_import COMMAND "${MIR3_GODOT_EXECUTABLE}" --headless --editor --path "${CMAKE_CURRENT_SOURCE_DIR}/godot" --import --quit)
            set_tests_properties(godot_import PROPERTIES FIXTURES_SETUP godot_project TIMEOUT 60)
            add_test(NAME hud_smoke COMMAND "${MIR3_GODOT_EXECUTABLE}" --headless --path "${CMAKE_CURRENT_SOURCE_DIR}/godot" --script "${CMAKE_CURRENT_SOURCE_DIR}/tests/hud_smoke.gd")
            add_test(NAME bichon_smoke COMMAND "${MIR3_GODOT_EXECUTABLE}" --headless --path "${CMAKE_CURRENT_SOURCE_DIR}/godot" --script "${CMAKE_CURRENT_SOURCE_DIR}/tests/hud_smoke.gd" -- --terrain)
            set_tests_properties(hud_smoke bichon_smoke PROPERTIES FIXTURES_REQUIRED godot_project TIMEOUT 45)
        endif()
    endif()
endif()
